    1. Create frontend/src/app/api/review/stream/route.ts:
       - Next.js API route for SSE proxy to backend
       - Connects to backend SSE for generation progress
       - Forwards events to frontend client

       ```typescript
       import { NextRequest } from 'next/server'

       export async function GET(request: NextRequest) {
         const { searchParams } = new URL(request.url)
         const clientId = searchParams.get('clientId')
         const token = searchParams.get('token')

         // Connect to backend SSE
         const backendUrl = `${process.env.NEXT_PUBLIC_API_URL}/api/generation/${clientId}/progress?token=${token}`

         const response = await fetch(backendUrl)

         // Forward SSE stream
         return new Response(response.body, {
           headers: {
             'Content-Type': 'text/event-stream',
             'Cache-Control': 'no-cache',
             'Connection': 'keep-alive',
           },
         })
       }
       ```
